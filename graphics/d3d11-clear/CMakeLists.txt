cmake_minimum_required(VERSION 3.21)
project(d3d11_clear CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This example uses Direct3D 11 and builds only for Windows.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  add_compile_options(/W4 /WX /permissive-)
else()
  add_compile_options(-Wall -Wextra -Werror)
endif()

add_executable(d3d11_clear WIN32 src/d3d11_clear.cpp)
target_link_libraries(d3d11_clear PRIVATE d3d11)

add_executable(warp_clear_test src/warp_clear_test.cpp)
target_link_libraries(warp_clear_test PRIVATE d3d11)
