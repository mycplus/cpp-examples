cmake_minimum_required(VERSION 3.21)
project(d3d12_device_info CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This example uses Direct3D 12 and builds only for Windows.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  add_compile_options(/W4 /WX /permissive-)
else()
  add_compile_options(-Wall -Wextra -Werror)
endif()

add_executable(d3d12_device_info src/d3d12_device_info.cpp)
target_link_libraries(d3d12_device_info PRIVATE d3d12 dxgi)
